CREATE TABLE IF NOT EXISTS users_profile (
  -- Column A: Primary Key (O(1) Generated)
  users_id    TEXT        NOT NULL PRIMARY KEY,
  -- Column B: Password (Plain or Hashed)
  password    TEXT        NOT NULL DEFAULT '',
  -- Column C: Full Name (Thai/English)
  users_name  TEXT        NOT NULL DEFAULT '',
  -- Column D: LINE UID (optional, from LINE LIFF)
  line_uid    TEXT        NULL DEFAULT '',
  -- Column E: Employee Number
  emp_no      TEXT        NULL DEFAULT '',
  -- Column F: Email
  email       TEXT        NULL DEFAULT '',
  -- Column G: Account Status ('Y' = Active, 'N' = Inactive)
  --           ค่า Default เมื่อสร้างใหม่ = 'N' (รอ Admin อนุมัติ)
  active      TEXT        NOT NULL DEFAULT 'N' CHECK (active IN ('Y', 'N')),
  -- Column H: Screen Authorization Tags (JSON Array stored as TEXT)
  --           ตัวอย่าง: '["PC01","PC02"]'
  screen_tags TEXT        NOT NULL DEFAULT '[]',
  -- Column I: Approver Tags (JSON Array stored as TEXT)
  --           ตัวอย่าง: '["คุมวงเงินสด","อนุมัติวงเงินสด"]'
  approve_tags TEXT       NOT NULL DEFAULT '[]',
  -- Audit fields
  created_at  TEXT        NOT NULL DEFAULT (datetime('now')),
  updated_at  TEXT        NOT NULL DEFAULT (datetime('now'))
);

-- Index สำหรับค้นหาด้วย line_uid (O(1) Lookup)
CREATE UNIQUE INDEX IF NOT EXISTS idx_users_profile_line_uid
  ON users_profile (line_uid)
  WHERE line_uid IS NOT NULL AND line_uid != '';

-- Index สำหรับ Query ผู้ใช้ Active
CREATE INDEX IF NOT EXISTS idx_users_profile_active
  ON users_profile (active);
```

วิธีใช้ กรณึต้องการ list approveList คนที่ ถือวงเงินสด   ในระบบบก็จะ config ว่าให้ค้นหา เฉพาะ "คุมวงเงินสด" 
หรือ ในหน้าจอ อนุมัติ การจ่ายชดเลยวงเงิน ก็จะ config ว่า หน้าจอนี้หให้ ค้นหาผู้อนุมัติ เฉพาะ "อนุมัติวงเงินสด" 
ซึ่ง ขึ้นกลับ request ของ งาน ในแต่ ละ webapp ว่าต้องการ ค้นหา กรอง filter ผู้อนุมัติ นั้นๆ ด้วยอะไร
